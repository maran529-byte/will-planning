### 3.4 文书生成器改造 (零成本)

现有 `t4-contract-generator/` 新增一个 `jurisdiction` 字段:

```typescript
// packages/contracts/types.ts
export type DocType = 'prenup' | 'postnup' | 'divorce' | 'custody' | 'gift' | 'will';
export type Locale = 'zh-CN' | 'en-US';

// 涉外标志, 不分国籍, 只问"是否有境外因素"
export interface CrossBorderExtras {
  hasForeignAsset: boolean;       // 是否有境外财产
  hasForeignParty: boolean;       // 对方是否外籍/绿卡
  hasForeignResidency: boolean;   // 任一方是否境外居住 > 180 天/年
  foreignJurisdictions: string[]; // ['US-CA', 'UK', 'SG', ...] (参考用)
  applicableLawHint: string;      // 自动生成 "准据法: 中华人民共和国法律"
}

export interface ContractInput {
  docType: DocType;
  locale: Locale;
  crossBorder?: CrossBorderExtras; // 可选, 启用时附涉外附件
  parties: Party[];
  assets: Asset[];
  children?: Child[];
}
```

**生成逻辑**:
```typescript
// packages/contracts/generator.ts
export function generateContract(input: ContractInput): Buffer {
  const baseTemplate = loadTemplate(input.docType);  // 现有中文模板
  const baseDoc = render(baseTemplate, input);

  if (input.crossBorder && hasAnyForeignFactor(input.crossBorder)) {
    const crossBorderAttachment = loadTemplate('cross_border_attachment', input.docType);
    return merge(baseDoc, render(crossBorderAttachment, input.crossBorder));
  }
  return baseDoc;
}
```

> **0 新增模板** — 仅在现有 6 类模板尾部追加"涉外附件"。
